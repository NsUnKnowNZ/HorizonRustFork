function Scene {
  $l = Select-String -Path $log -Pattern 'cinematic_interactions\] scene at panel (-?\d+) (-?\d+) book (-?\d+) (-?\d+) result (-?\d+) (-?\d+) layout (-?\d+) (-?\d+) cells (.*)$' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $script:px=[int]$g[1].Value; $script:py=[int]$g[2].Value; $script:bx=[int]$g[3].Value; $script:by=[int]$g[4].Value; $script:rx=[int]$g[5].Value; $script:ry=[int]$g[6].Value; $script:lx=[int]$g[7].Value; $script:ly=[int]$g[8].Value
  $script:cells = @($g[9].Value.Trim().Split(' ') | ForEach-Object { [int]$_ })
  "scene: $($l.Line)"
}
function Station {
  $l = Select-String -Path $log -Pattern 'cinematic_interactions\] station (\w+) at panel (-?\d+) (-?\d+) spots (.*)$' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $script:px=[int]$g[2].Value; $script:py=[int]$g[3].Value
  $script:spots = @($g[4].Value.Trim().Split(' ') | ForEach-Object { [int]$_ })
  "station: $($l.Line)"
}
function CellX($i) { return $script:cells[2*$i] }
function CellY($i) { return $script:cells[2*$i+1] }
function SpotX($i) { return $script:spots[2*$i] }
function SpotY($i) { return $script:spots[2*$i+1] }
function HotX($i) { return $script:px + 13 + 18*$i }
function HotY { return $script:py + 71 }
function InvX($k) { return $script:px + 13 + 18*($k % 9) }
function InvY($k) { return $script:py + 13 + 18*[math]::Floor($k / 9) }
function Open($x, $y, $z) {
  C command "tp @s $($x + 0.5) 150 207.3 180 35"
  C wait 10
  C look-at-block $x $y $z south
  C mouse right click
  C wait 30
}
