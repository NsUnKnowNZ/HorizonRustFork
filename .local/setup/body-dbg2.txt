$log = "$env:TEMP\horizon-native-test.log"
function Pack($pack, $tag) {
  "=== pack: $pack"
  $env:HORIZON_SHADER_PACK = $pack
  C launch --world NativeTest --wait --offscreen
  for ($i = 0; $i -lt 120; $i++) { $r = & $ctl --socket "$env:TEMP\horizon-native-test.json" --timeout 10 status 2>&1 | Out-String; if ($r -match '"joined":true|"chunks"') { break }; Start-Sleep 5 }
  Get-CimInstance Win32_Process -Filter "Name='horizon-client.exe'" | Where-Object { $_.CommandLine -like "*horizon-native-test*" } | ForEach-Object { (Get-Process -Id $_.ProcessId).PriorityClass = "Normal" }
  C wait 20
  Platform
  C command "gamemode survival"
  C command "kill @e[type=!player]"
  C command "clear @s"
  C command "setblock 204 150 205 minecraft:crafting_table"
  C command "summon minecraft:text_display 203.0 151.6 205.5 {text:{text:'Break the block!',color:'gold'},billboard:'center',shadow:1,alignment:'center'}"
  C command "summon minecraft:item 206.5 150.2 205.5 {Item:{id:'minecraft:diamond',count:1},PickupDelay:32767}"
  C command "item replace entity @s hotbar.0 with minecraft:oak_planks 1"
  C command "item replace entity @s hotbar.1 with minecraft:oak_planks 1"
  C command "item replace entity @s hotbar.2 with minecraft:oak_planks 1"
  C command "tp @s 204.5 150 207.3 180 35"
  C wait 10
  C look-at-block 204 150 205 south
  C mouse right click
  C wait 30
  $l = Select-String -Path $log -Pattern 'cinematic_interactions\] scene at panel (-?\d+) (-?\d+) .* cells (.*)$' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $px=[int]$g[1].Value; $py=[int]$g[2].Value
  $cells = @($g[3].Value.Trim().Split(' ', [System.StringSplitOptions]::RemoveEmptyEntries) | ForEach-Object { [int]$_ })
  foreach ($i in 0,1,2) { C click-at ($px + 13 + 18*$i) ($py + 71); C click-at $cells[2*$i] $cells[2*$i+1] }
  C wait 20
  C screenshot "$shots\sp-$tag-scene.png"
  C key e tap
  C wait 20
  C command "tp @s 204.5 150 208.5 180 20"
  C wait 20
  C screenshot "$shots\sp-$tag-world.png"
  C quit
  Start-Sleep 6
  Get-CimInstance Win32_Process -Filter "Name='horizon-client.exe'" | Where-Object { $_.CommandLine -like "*horizon-native-test*" } | ForEach-Object { Stop-Process -Id $_.ProcessId -Force }
}
Pack 'Bliss_v2.1.2_(Chocapic13_Shaders_edit).zip' 'bliss'
function Run($pack, $tag) {
  "=== pack: $pack"
  $env:HORIZON_SHADER_PACK = $pack
  C launch --world NativeTest --wait --offscreen
  for ($i = 0; $i -lt 120; $i++) { $r = & $ctl --socket "$env:TEMP\horizon-native-test.json" --timeout 10 status 2>&1 | Out-String; if ($r -match '"joined":true|"chunks"') { break }; Start-Sleep 5 }
  Get-CimInstance Win32_Process -Filter "Name='horizon-client.exe'" | Where-Object { $_.CommandLine -like "*horizon-native-test*" } | ForEach-Object { (Get-Process -Id $_.ProcessId).PriorityClass = "Normal" }
  C wait 20
  Platform
  C command "gamemode creative"
  C command "kill @e[type=!player]"
  C command "scoreboard objectives add tt dummy"
  C command "scoreboard players set #x tt 7"
  C command "summon minecraft:text_display 202.0 151.6 205.5 {text:{score:{name:'#x',objective:'tt'},color:'red',bold:true},billboard:'center',alignment:'center'}"
  C command "summon minecraft:text_display 204.5 151.6 205.5 {Tags:['st'],text:{score:{name:'#x',objective:'tt'},color:'red',bold:true},billboard:'center',see_through:1,alignment:'center'}"
  C command "summon minecraft:text_display 207.0 151.6 205.5 {text:{text:'plain',color:'gold'},billboard:'center',see_through:1}"
  C command "summon minecraft:text_display 204.5 152.4 205.5 {text:{text:'Upgrade in',color:'white',bold:true},billboard:'center',shadow:1,alignment:'center'}"
  C command "summon minecraft:item 206.5 150.2 207.0 {Item:{id:'minecraft:diamond',count:1},PickupDelay:32767}"
  C command "tp @s 204.5 150 209.5 180 -5"
  C wait 30
  C screenshot "$shots\dbg-$tag-1.png"
  C command "data get entity @e[tag=st,limit=1] text"
  C command "scoreboard players set #x tt 3"
  C command "data modify entity @e[tag=st,limit=1] text set value {score:{name:'#x',objective:'tt'},color:'gold',bold:true}"
  C wait 30
  C screenshot "$shots\dbg-$tag-2.png"
  C quit
  Start-Sleep 6
  Get-CimInstance Win32_Process -Filter "Name='horizon-client.exe'" | Where-Object { $_.CommandLine -like "*horizon-native-test*" } | ForEach-Object { Stop-Process -Id $_.ProcessId -Force }
  Copy-Item $log "$shots\dbg-$tag.log" -Force
}
