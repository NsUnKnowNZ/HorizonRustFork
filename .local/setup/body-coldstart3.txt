function Times($tag) {
  $l = & $ctl --socket "$env:TEMP\horizon-native-test.json" --timeout 30 logs --lines 4000 2>&1 | Out-String
  "--- $tag"
  [regex]::Matches($l, '(load core tables: [0-9.]+s|load items: [0-9.]+s|resources loaded in [0-9.]+s|mod files checked in \d+ ms|wrote horizon-data.bin in \d+ ms|horizon-data.bin not written[^"\\\r\n]*|content from \d+ mod)') | ForEach-Object { $_.Value } | Select-Object -Unique | Select-Object -Last 10
}
