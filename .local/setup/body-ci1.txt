function Scene {
  $l = Select-String -Path $log -Pattern 'cinematic_interactions\] scene at panel (-?\d+) (-?\d+) book (-?\d+) (-?\d+) result (-?\d+) (-?\d+) layout (-?\d+) (-?\d+) cells (.*)$' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $script:px=[int]$g[1].Value; $script:py=[int]$g[2].Value; $script:bx=[int]$g[3].Value; $script:by=[int]$g[4].Value; $script:rx=[int]$g[5].Value; $script:ry=[int]$g[6].Value; $script:lx=[int]$g[7].Value; $script:ly=[int]$g[8].Value
  $script:cells = @($g[9].Value.Trim().Split(' ') | ForEach-Object { [int]$_ })
  "scene: $($l.Line)"
}
function CellX($i) { return $script:cells[2*$i] }
function CellY($i) { return $script:cells[2*$i+1] }
function HotX($i) { return $script:px + 13 + 18*$i }
function HotY { return $script:py + 71 }
