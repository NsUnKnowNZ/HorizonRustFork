function Scene {
  $l = Select-String -Path "$env:TEMP\horizon-native-test.log" -Pattern 'scene at strip (-?\d+) (-?\d+) card (-?\d+) (-?\d+) middle (-?\d+) (-?\d+)' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $script:sx=[int]$g[1].Value; $script:sy=[int]$g[2].Value; $script:cx=[int]$g[3].Value; $script:cy=[int]$g[4].Value; $script:mx=[int]$g[5].Value; $script:my=[int]$g[6].Value
  "scene: $($l.Line)"
}
