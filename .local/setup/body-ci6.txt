function LaunchOff {
  C launch --world NativeTest --wait --offscreen
  for ($i = 0; $i -lt 120; $i++) { $r = & $ctl --socket "$env:TEMP\horizon-native-test.json" --timeout 10 status 2>&1 | Out-String; if ($r -match '"joined":true|"chunks"') { break }; Start-Sleep 5 }
  "ready after $($i*5)s more"
  Get-CimInstance Win32_Process -Filter "Name='horizon-client.exe'" | Where-Object { $_.CommandLine -like "*horizon-native-test*" } | ForEach-Object { (Get-Process -Id $_.ProcessId).PriorityClass = "Normal" }
  C wait 20
  C respawn
  C wait 20
}

function Station {
  $l = Select-String -Path $log -Pattern 'cinematic_interactions\] station (\w+) at panel (-?\d+) (-?\d+) result (-?\d+) (-?\d+) spots(.*)$' | Select-Object -Last 1
  $g = $l.Matches[0].Groups
  $script:px=[int]$g[2].Value; $script:py=[int]$g[3].Value; $script:rx=[int]$g[4].Value; $script:ry=[int]$g[5].Value
  $script:spots = @($g[6].Value.Trim().Split(' ', [System.StringSplitOptions]::RemoveEmptyEntries) | ForEach-Object { [int]$_ })
  "station: $($l.Line)"
}
function SpotX($i) { return $script:spots[2*$i] }
function SpotY($i) { return $script:spots[2*$i+1] }
function HotX($i) { return $script:px + 13 + 18*$i }
function HotY { return $script:py + 71 }
function Open($x, $face) {
  C command "tp @s $($x + 0.5) 150 207.3 180 35"
  C wait 10
  C look-at-block $x 150 205 $face
  C mouse right click
  C wait 30
  Station
}
function Lay($slotX, $slotY, $spot) { C click-at $slotX $slotY; C click-at (SpotX $spot) (SpotY $spot) }
